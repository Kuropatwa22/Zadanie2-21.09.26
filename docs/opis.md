## Przycisk środkowy (button)

layout_width = ustawia szerokosc guzika.

layout_height = ustawia wysokosc guzika.

id = nadaje guzikowi jego nazwe.

text = ustawia napis na guziku.

constraintBottom_toBottomOf = przyczepia dol guzika do dolu ekranu.
constraintEnd_toEndOf = przyczepia prawa strone guzika do prawej strony ekranu.
constraintHorizontal_bias = ustawia guzik mniej wiecej na srodku w poziomie.
constraintStart_toStartOf = przyczepia lewa strone guzika do lewej strony ekranu.
constraintTop_toTopOf = przyczepia gore guzika do gory ekranu.
constraintVertical_bias = ustawia guzik mniej wiecej na srodku w pionie.

**Przycisk z biasem (button6)**

id = nadaje guzikowi jego nazwe.
layout_width = ustawia szerokosc guzika.
layout_height = ustawia wysokosc guzika.
text = ustawia napis na guziku.
constraintBottom_toBottomOf = przyczepia dol guzika do dolu ekranu.
constraintTop_toBottomOf = ustawia gore guzika wzgledem dolu srodkowego guzika.
constraintVertical_bias = ustawia guzik mniej wiecej w polowie miejsca miedzy srodkowym guzikiem a dolem ekranu.
tools:layout_editor_absoluteX = ustawia pozycje guzika w edytorze, ale nie podczas dzialania aplikacji.
